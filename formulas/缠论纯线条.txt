{缠论纯线条 v6.2}
{主图叠加指标 - 仅显示笔和中枢矩形，无图标}

{==== 获取数据 ====}
BI:=TDXDLL1(2, H, L, C);
ZSZG:=TDXDLL1(3, H, L, C);
ZSZD:=TDXDLL1(4, H, L, C);
ZSKS:=TDXDLL1(18, H, L, C);
ZSJS:=TDXDLL1(19, H, L, C);
KXG:=TDXDLL1(20, H, L, C);
KXD:=TDXDLL1(21, H, L, C);

{==== 笔连线 ====}
DRAWLINE(BI=1, KXG, BI=-1, KXD, 0), COLORYELLOW, LINETHICK2;
DRAWLINE(BI=-1, KXD, BI=1, KXG, 0), COLORYELLOW, LINETHICK2;

{==== 中枢矩形 ====}
{下跌中枢 - 绿色}
DRAWLINE(ZSKS=1, ZSZG, ZSJS=1, ZSZG, 0), COLOR00AA00, LINETHICK1;
DRAWLINE(ZSKS=1, ZSZD, ZSJS=1, ZSZD, 0), COLOR00AA00, LINETHICK1;
STICKLINE(ZSKS=1, ZSZG, ZSZD, 0.1, 0), COLOR00AA00;
STICKLINE(ZSJS=1, ZSZG, ZSZD, 0.1, 0), COLOR00AA00;

{上涨中枢 - 红色}
DRAWLINE(ZSKS=2, ZSZG, ZSJS=2, ZSZG, 0), COLORRED, LINETHICK1;
DRAWLINE(ZSKS=2, ZSZD, ZSJS=2, ZSZD, 0), COLORRED, LINETHICK1;
STICKLINE(ZSKS=2, ZSZG, ZSZD, 0.1, 0), COLORRED;
STICKLINE(ZSJS=2, ZSZG, ZSZD, 0.1, 0), COLORRED;
