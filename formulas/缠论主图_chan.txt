{缠论主图 v7.5 - 适配 chan.dll (TDXDLL1)}
{参照原版缠论主图.txt，使用我们的DLL接口编号}

{==== 核心数据获取 ====}
{BI: 笔类型 1=顶端点, -1=底端点, 0=非端点}
BI:=TDXDLL1(2, H, L, C);
{KXG: 笔高点价格（只在顶点有值）}
KXG:=TDXDLL1(20, H, L, C);
{KXD: 笔低点价格（只在底点有值）}
KXD:=TDXDLL1(21, H, L, C);
{中枢数据}
ZSZG:=TDXDLL1(3, H, L, C);
ZSZD:=TDXDLL1(4, H, L, C);
ZSKS:=TDXDLL1(18, H, L, C);
ZSJS:=TDXDLL1(19, H, L, C);

{==== 画笔 ====}
DRAWLINE(BI=1, KXG, BI=-1, KXD, 0), COLORYELLOW;
DRAWLINE(BI=-1, KXD, BI=1, KXG, 0), COLORYELLOW;

{==== 下跌笔中枢（绿色）====}
DRAWLINE(ZSKS=1, ZSZG, ZSJS=1, ZSZG, 0), COLOR1D7300, LINETHICK2;
DRAWLINE(ZSKS=1, ZSZD, ZSJS=1, ZSZD, 0), COLOR1D7300, LINETHICK2;
DRAWLINE(ZSKS=1, (ZSZG+ZSZD)/2, ZSJS=1, (ZSZG+ZSZD)/2, 0), COLOR1D7300, DOTLINE;
STICKLINE(ZSKS=1, ZSZG, ZSZD, 0.1, 0), COLOR1D7300;
STICKLINE(ZSJS=1, ZSZG, ZSZD, 0.1, 0), COLOR1D7300;

{==== 上涨笔中枢（红色）====}
DRAWLINE(ZSKS=2, ZSZG, ZSJS=2, ZSZG, 0), COLORRED, LINETHICK2;
DRAWLINE(ZSKS=2, ZSZD, ZSJS=2, ZSZD, 0), COLORRED, LINETHICK2;
DRAWLINE(ZSKS=2, (ZSZG+ZSZD)/2, ZSJS=2, (ZSZG+ZSZD)/2, 0), COLORRED, DOTLINE;
STICKLINE(ZSKS=2, ZSZG, ZSZD, 0.1, 0), COLORRED;
STICKLINE(ZSJS=2, ZSZG, ZSZD, 0.1, 0), COLORRED;

{==== 计算笔距离（公式层计算，无需DLL）====}
H1:=BARSLAST(BI=1);
L1:=BARSLAST(BI=-1);
HH1:=REF(H1+1, 1);
LL1:=REF(L1+1, 1);

{==== 计算方向 ====}
A1:=BARSLAST(BI<>0);
A2:=REF(BI, A1);
方向:=-REF(A2, 1);

{==== 高低点序列（GG/DD）====}
GG:=REF(H, H1);
GG1:=REF(H, HH1);
GG2:=REF(GG1, HH1);
GG3:=REF(GG2, HH1);
GG4:=REF(GG3, HH1);
GG5:=REF(GG4, HH1);

DD:=REF(L, L1);
DD1:=REF(L, LL1);
DD2:=REF(DD1, LL1);
DD3:=REF(DD2, LL1);
DD4:=REF(DD3, LL1);
DD5:=REF(DD4, LL1);

{==== 均线 ====}
MA26:=EMA(C, 26);
MA13:=EMA(C, 13);

{==== 买点逻辑（完整按手册）====}
{一买}
一买TJ1:=方向=1 AND L<MA13 AND LL1<=5;
一五段下跌:=DD1<GG1 AND DD1<DD2 AND DD1<DD3 AND GG1<GG2 AND GG1<GG3;
一买TJA:=GG1<DD3;
一买A:=一买TJ1 AND 一五段下跌 AND 一买TJA, LINETHICK0;
一买KJA:=GG1-DD1<GG2-DD2 AND GG1-DD1>GG3-DD3 AND GG3-DD3<GG2-DD2 AND GG2-DD2>(GG1-DD1)*1.618;
一买AAA:=一买TJ1 AND 一五段下跌 AND 一买TJA AND 一买KJA, LINETHICK0;
一买TJB:=GG1>DD3;
一买KJB:=GG3-DD3>GG1-DD1 AND GG3-DD3>GG2-DD2 AND GG2-DD2<GG1-DD1;
一买B:=一买TJ1 AND 一五段下跌 AND 一买TJB AND 一买KJB, LINETHICK0;

{类二买}
类二买TJ:=方向=1 AND C<MA13 AND LL1<=8;
类二买TJ2:=DD1<GG1 AND DD3<DD2 AND DD3<DD1 AND DD3<DD4;
类二买KJ:=GG2-DD3>GG2-DD2 AND GG2-DD3>GG1-DD1;
类二买A:=类二买TJ AND 类二买TJ2 AND 类二买KJ, LINETHICK0;
类二买TJA:=GG1>GG2 AND DD1>DD2 AND GG2-DD3>(GG2-DD2)*1.618;
类二买AAA:=类二买A AND 类二买TJA, LINETHICK0;

{二买}
买TJ1:=方向=1 AND L<MA26 AND LL1<=8;
二买TJ:=DD1<GG1 AND DD1>DD2;
三段下跌:=GG3>GG2 AND DD3>DD2;
二买TJA1:=GG1>DD3;
二买A:=买TJ1 AND 二买TJ AND 三段下跌 AND 二买TJA1, LINETHICK0;
五段下跌:=GG4>GG3 AND GG4>GG2 AND DD2<DD3 AND DD2<DD4;
二买TJB1:=GG2<DD4 AND GG1>DD3;
二买TJB2:=GG2>DD4;
二买B1:=买TJ1 AND 二买TJ AND 五段下跌 AND 二买TJB1, LINETHICK0;
二买B2:=买TJ1 AND 二买TJ AND 五段下跌 AND 二买TJB2, LINETHICK0;
二买AAA:=二买B1, LINETHICK0;

{三买}
三买TJ:=DD1<GG1 AND DD1>DD2;
三买TJA1:=方向=1 AND L<MA13 AND LL1<=5;
三买TJA2:=DD1>MIN(GG2,GG3) AND GG3>DD2 AND DD4<MAX(DD2,DD3) AND DD1>DD4;
三买A:=三买TJ AND 三买TJA1 AND 三买TJA2, LINETHICK0;

{==== 卖点逻辑 ====}
{一卖}
一卖TJ1:=方向=-1 AND H>MA13 AND HH1<=5;
一五段上涨:=GG1>GG2 AND GG1>GG3 AND DD1>DD2 AND DD1>DD3;
一卖TJA:=DD1>GG3;
一卖A:=一卖TJ1 AND 一五段上涨 AND 一卖TJA, LINETHICK0;
一卖TJB:=DD1<GG3 AND GG1-DD1>GG2-DD2 AND GG3-DD3-GG2-DD2;
一卖B:=一卖TJ1 AND 一五段上涨 AND 一卖TJB, LINETHICK0;
一卖TJC:=GG1>GG2 AND GG2>GG3 AND GG3>GG4 AND GG4>GG5 AND DD1>DD2 AND DD2>DD3 AND DD3>DD4 AND DD4>DD5;
一卖C:=一卖TJ1 AND 一卖TJC, LINETHICK0;

{二卖}
卖TJ1:=方向=-1 AND H>MA13 AND HH1<=8;
二卖TJ:=GG1>DD1 AND GG1<GG2;
三段上涨:=GG3<GG2 AND DD3<DD2;
二卖TJA1:=DD1>GG3;
二卖A:=卖TJ1 AND 二卖TJ AND 三段上涨 AND 二卖TJA1, LINETHICK0;
五段上涨:=GG2>GG3 AND GG2>GG4 AND DD4<DD3 AND DD4<DD2;
二卖TJB1:=GG4<DD2 AND DD1<GG3;
二卖TJB2:=GG4>DD2;
二卖B1:=卖TJ1 AND 二卖TJ AND 五段上涨 AND 二卖TJB1, LINETHICK0;
二卖B2:=卖TJ1 AND 二卖TJ AND 五段上涨 AND 二卖TJB2, LINETHICK0;
一段上涨:=GG2>GG3 AND GG2>GG4 AND DD1>MIN(GG3, GG4);
二卖C1:=卖TJ1 AND 二卖TJ AND 一段上涨, LINETHICK0;

{三卖}
三卖TJ:=DD1<GG1 AND GG1<GG2;
三卖TJA1:=方向=-1 AND H>MA13 AND HH1<=5;
三卖TJA2:=GG1<MAX(DD2,DD3) AND GG2>DD3 AND GG4>MIN(GG2,GG3) AND GG1<GG4;
三卖A:=三卖TJ AND 三卖TJA1 AND 三卖TJA2, LINETHICK0;

{==== 综合买卖点 ====}
一买:=一买A OR 一买B;
类二买:=类二买A;
二买:=二买A OR 二买B1 OR 二买B2;
三买:=三买A;

一卖:=一卖A OR 一卖B OR 一卖C;
二卖:=二卖A OR 二卖B1 OR 二卖B2 OR 二卖C1;
三卖:=三卖A;

{==== 买卖点提示 ====}
DRAWTEXT(一买 AND FILTER(一买, 8), REF(L,1), '①'), COLORYELLOW;
DRAWTEXT(类二买 AND FILTER(类二买, 8), REF(L,1), '类②'), COLORYELLOW;
DRAWTEXT(二买 AND FILTER(二买, 8), REF(L,1), '②'), COLORYELLOW;
DRAWTEXT(三买 AND FILTER(三买, 8), REF(L,1), '③'), COLORYELLOW;
DRAWTEXT(一卖 AND FILTER(一卖, 8), REF(H,1), '①'), COLORGREEN;
DRAWTEXT(二卖 AND FILTER(二卖, 8), REF(H,1), '②'), COLORGREEN;
DRAWTEXT(三卖 AND FILTER(三卖, 8), REF(H,1), '③'), COLORGREEN;

{==== 涨停板 ====}
TB:=(C-REF(C,1))/REF(C,1)*100;
STICKLINE(TB>9.9, O, C, 2.5, 1), COLORMAGENTA;
STICKLINE(TB<-9.9, O, C, 2.5, 0), COLOR1D7300;
