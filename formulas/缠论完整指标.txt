{缠论完整指标 v7.5.1}
{主图叠加指标 - 包含笔、中枢、买卖点}
{匹配模型: 最新匹配 — DLL信号 + BACKSET回写到笔端点bar}

{==== 函数编号说明 ====}
{编号2: BI - 笔端点类型 (1=顶, -1=底, 0=非端点)}
{编号3: ZSZG - 中枢高点（持续输出）}
{编号4: ZSZD - 中枢低点（持续输出）}
{编号7: BSIG - 买点信号}
{编号8: SSIG - 卖点信号}
{编号20: KXG - 顶端点价格}
{编号21: KXD - 底端点价格}
{编号22: ZS_DIR - 中枢方向（持续输出）}

{==== 获取数据 ====}
BI:=TDXDLL1(2, H, L, C);
KXG:=TDXDLL1(20, H, L, C);
KXD:=TDXDLL1(21, H, L, C);
ZSZG:=TDXDLL1(3, H, L, C);
ZSZD:=TDXDLL1(4, H, L, C);
BSIG:=TDXDLL1(7, H, L, C);
SSIG:=TDXDLL1(8, H, L, C);

{==== 买卖点锚定位置 ====}
BPOS:=IF(KXD>0, KXD*0.998, L);
SPOS:=IF(KXG>0, KXG*1.002, H);

{==== 笔连线 ====}
DRAWLINE(BI=1, KXG, BI=-1, KXD, 0), COLORYELLOW;
DRAWLINE(BI=-1, KXD, BI=1, KXG, 0), COLORYELLOW;

{==== 中枢区域（边框画法）====}
在中枢内:=ZSZG>ZSZD;
ZS_DIR:=TDXDLL1(22, H, L, C);
ZSKS:=TDXDLL1(18, H, L, C);
ZSJS:=TDXDLL1(19, H, L, C);
ZZ:=(ZSZG+ZSZD)/2;

{上涨中枢边框 — 红色}
STICKLINE(在中枢内 AND ZS_DIR>0, ZSZG, ZSZG, 0.2, 0), COLORRED;
STICKLINE(在中枢内 AND ZS_DIR>0, ZSZD, ZSZD, 0.2, 0), COLORRED;
STICKLINE(ZSKS=2, ZSZG, ZSZD, 0.1, 0), COLORRED;
STICKLINE(ZSJS=2, ZSZG, ZSZD, 0.1, 0), COLORRED;

{下跌中枢边框 — 绿色}
STICKLINE(在中枢内 AND ZS_DIR<0, ZSZG, ZSZG, 0.2, 0), COLOR1D7300;
STICKLINE(在中枢内 AND ZS_DIR<0, ZSZD, ZSZD, 0.2, 0), COLOR1D7300;
STICKLINE(ZSKS=1, ZSZG, ZSZD, 0.1, 0), COLOR1D7300;
STICKLINE(ZSJS=1, ZSZG, ZSZD, 0.1, 0), COLOR1D7300;

{中轴虚线}
STICKLINE(在中枢内, ZZ, ZZ, 0.1, 0), COLOR808080;

{==== BACKSET回写到笔端点 ====}
BOTN:=BARSLAST(BI=-1);
TOPN:=BARSLAST(BI=1);

B1A:=BACKSET(BSIG=1,  BOTN+1) AND BI=-1;
B1B:=BACKSET(BSIG=2,  BOTN+1) AND BI=-1;
B2A:=BACKSET(BSIG=11, BOTN+1) AND BI=-1;
B2B1:=BACKSET(BSIG=12, BOTN+1) AND BI=-1;
B2B2:=BACKSET(BSIG=13, BOTN+1) AND BI=-1;
B3:=BACKSET(BSIG=21,  BOTN+1) AND BI=-1;

S1A:=BACKSET(SSIG=-1,  TOPN+1) AND BI=1;
S1B:=BACKSET(SSIG=-2,  TOPN+1) AND BI=1;
S1C:=BACKSET(SSIG=-3,  TOPN+1) AND BI=1;
S2A:=BACKSET(SSIG=-11, TOPN+1) AND BI=1;
S2B1:=BACKSET(SSIG=-12, TOPN+1) AND BI=1;
S2B2:=BACKSET(SSIG=-13, TOPN+1) AND BI=1;
S3:=BACKSET(SSIG=-21,  TOPN+1) AND BI=1;

{==== 买点信号（标注在笔底端点）====}
DRAWTEXT(B1A,  BPOS, 'B1A'),  COLORFF0000;
DRAWTEXT(B1B,  BPOS, 'B1B'),  COLORFF0000;
DRAWTEXT(B2A,  BPOS, 'B2A'),  COLOR0066FF;
DRAWTEXT(B2B1, BPOS, 'B2B1'), COLOR0066FF;
DRAWTEXT(B2B2, BPOS, 'B2B2'), COLOR0066FF;
DRAWTEXT(B3,   BPOS, 'B3'),   COLORFF00FF;

{==== 卖点信号（标注在笔顶端点）====}
DRAWTEXT(S1A,  SPOS, 'S1A'),  COLOR00CC00;
DRAWTEXT(S1B,  SPOS, 'S1B'),  COLOR00CC00;
DRAWTEXT(S1C,  SPOS, 'S1C'),  COLOR00CC00;
DRAWTEXT(S2A,  SPOS, 'S2A'),  COLORFFAA00;
DRAWTEXT(S2B1, SPOS, 'S2B1'), COLORFFAA00;
DRAWTEXT(S2B2, SPOS, 'S2B2'), COLORFFAA00;
DRAWTEXT(S3,   SPOS, 'S3'),   COLORWHITE;
