{缠论完整指标 v7.4}
{主图叠加指标 - 包含笔、中枢、买卖点}

{==== 函数编号说明 ====}
{编号2: BI - 笔端点类型 (1=顶, -1=底, 0=非端点)}
{编号3: ZSZG - 中枢高点（持续输出）}
{编号4: ZSZD - 中枢低点（持续输出）}
{编号7: BSIG - 买点信号}
{编号8: SSIG - 卖点信号}
{编号20: KXG - 顶端点价格}
{编号21: KXD - 底端点价格}
{编号22: ZS_DIR - 中枢方向（持续输出）}

{==== 获取数据 ====}
BI:=TDXDLL1(2, H, L, C);
KXG:=TDXDLL1(20, H, L, C);
KXD:=TDXDLL1(21, H, L, C);
ZSZG:=TDXDLL1(3, H, L, C);
ZSZD:=TDXDLL1(4, H, L, C);
BSIG:=TDXDLL1(7, H, L, C);
SSIG:=TDXDLL1(8, H, L, C);

{==== 笔连线 ====}
DRAWLINE(BI=1, KXG, BI=-1, KXD, 0), COLORYELLOW;
DRAWLINE(BI=-1, KXD, BI=1, KXG, 0), COLORYELLOW;

{==== 中枢区域 ====}
在中枢内:=ZSZG>ZSZD;
STICKLINE(在中枢内, ZSZG, ZSZD, 4, 1), COLORGRAY;

{==== 买点信号 ====}
DRAWTEXT(BSIG=1, L*0.98, 'B1A'), COLORFF0000;
DRAWTEXT(BSIG=2, L*0.98, 'B1B'), COLORFF0000;
DRAWTEXT(BSIG=11, L*0.97, 'B2A'), COLOR0066FF;
DRAWTEXT(BSIG=12, L*0.97, 'B2B1'), COLOR0066FF;
DRAWTEXT(BSIG=13, L*0.97, 'B2B2'), COLOR0066FF;
DRAWTEXT(BSIG=21, L*0.96, 'B3'), COLORFF00FF;

{==== 卖点信号 ====}
DRAWTEXT(SSIG=-1, H*1.02, 'S1A'), COLOR00CC00;
DRAWTEXT(SSIG=-2, H*1.02, 'S1B'), COLOR00CC00;
DRAWTEXT(SSIG=-3, H*1.02, 'S1C'), COLOR00CC00;
DRAWTEXT(SSIG=-11, H*1.03, 'S2A'), COLORFFAA00;
DRAWTEXT(SSIG=-12, H*1.03, 'S2B1'), COLORFFAA00;
DRAWTEXT(SSIG=-13, H*1.03, 'S2B2'), COLORFFAA00;
DRAWTEXT(SSIG=-21, H*1.04, 'S3'), COLORWHITE;
