{==== 获取买卖点数据 ====}
BSIG:=TDXDLL1(7, H, L, C);
SSIG:=TDXDLL1(8, H, L, C);
KXG:=TDXDLL1(20, H, L, C);
KXD:=TDXDLL1(21, H, L, C);

{==== 买卖点锚定位置 ====}
BPOS:=IF(KXD>0, KXD*0.998, L);
SPOS:=IF(KXG>0, KXG*1.002, H);

{==== 买点信号（锚定到笔底端点）====}
DRAWTEXT(BSIG=1, BPOS, 'B1A'), COLORFF0000;
DRAWTEXT(BSIG=2, BPOS, 'B1B'), COLORFF0000;
DRAWTEXT(BSIG=11, BPOS, 'B2A'), COLOR0066FF;
DRAWTEXT(BSIG=12, BPOS, 'B2B1'), COLOR0066FF;
DRAWTEXT(BSIG=13, BPOS, 'B2B2'), COLOR0066FF;
DRAWTEXT(BSIG=21, BPOS, 'B3'), COLORFF00FF;
DRAWTEXT(BSIG=31, BPOS, 'PB1'), COLORGRAY;
DRAWTEXT(BSIG=32, BPOS, 'PB2'), COLORGRAY;
DRAWTEXT(BSIG=33, BPOS, 'PB3'), COLORGRAY;

{==== 卖点信号（锚定到笔顶端点）====}
DRAWTEXT(SSIG=-1, SPOS, 'S1A'), COLOR00CC00;
DRAWTEXT(SSIG=-2, SPOS, 'S1B'), COLOR00CC00;
DRAWTEXT(SSIG=-3, SPOS, 'S1C'), COLOR00CC00;
DRAWTEXT(SSIG=-11, SPOS, 'S2A'), COLORFFAA00;
DRAWTEXT(SSIG=-12, SPOS, 'S2B1'), COLORFFAA00;
DRAWTEXT(SSIG=-13, SPOS, 'S2B2'), COLORFFAA00;
DRAWTEXT(SSIG=-21, SPOS, 'S3'), COLORWHITE;
DRAWTEXT(SSIG=-31, SPOS, 'PS1'), COLORGRAY;
DRAWTEXT(SSIG=-32, SPOS, 'PS2'), COLORGRAY;
DRAWTEXT(SSIG=-33, SPOS, 'PS3'), COLORGRAY;
