{缠论调试 - 用于检查DLL输出}

{==== 获取数据 ====}
FX:=TDXDLL1(1, H, L, C);   {分型: 1=顶, -1=底}
BI:=TDXDLL1(2, H, L, C);   {笔端点: 1=顶, -1=底}
KXG:=TDXDLL1(20, H, L, C); {笔高点价格}
KXD:=TDXDLL1(21, H, L, C); {笔低点价格}

{==== 显示原始输出值 ====}
分型:FX, NODRAW;
笔端点:BI, NODRAW;
高点价:KXG, NODRAW;
低点价:KXD, NODRAW;

{==== 显示条件判断 ====}
顶点位:IF(BI=1, 1, 0), NODRAW;
底点位:IF(BI=-1, 1, 0), NODRAW;
顶有价:IF(BI=1 AND KXG>0, 1, 0), NODRAW;
底有价:IF(BI=-1 AND KXD>0, 1, 0), NODRAW;

{==== 可视化标记 ====}
DRAWICON(FX=1, H, 23);  {顶分型-粉色三角}
DRAWICON(FX=-1, L, 24); {底分型-粉色三角}
DRAWICON(BI=1, H*1.01, 1);  {顶端点-向下箭头}
DRAWICON(BI=-1, L*0.99, 2); {底端点-向上箭头}

{==== 尝试画线 ====}
{如果BI=1和BI=-1对应位置都有正确价格，线应该正确连接}
DRAWLINE(BI=1, KXG, BI=-1, KXD, 0), COLORRED;
DRAWLINE(BI=-1, KXD, BI=1, KXG, 0), COLORRED;

{也尝试用原始价格}
DRAWLINE(BI=1, H, BI=-1, L, 0), COLORGREEN;
DRAWLINE(BI=-1, L, BI=1, H, 0), COLORGREEN;
